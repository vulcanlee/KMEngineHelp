提示詞清單（Prompt）管理頁面 Implementation Plan

Archived historical implementation plan: 本檔保留當時 agent 執行計畫與脈絡；下列步驟不再作為目前 backlog 追蹤。

Goal: 新增一個「提示詞清單」CRUD 管理頁面，位於側邊選單「資料定義」群組，供使用者集中維護可重複使用的 LLM 提示詞。

Architecture: 與既有「分類清單(Category)」同構的垂直切片（Entity → Repository → Service → Controller → DTO → AdapterModel → Blazor View），唯一資料差異是多一個「提示詞本文」(Content) 長文字欄位。不套用團隊權控（與分類/團隊清單一致，屬全系統共用資料定義）。預設依 UpdatedAt 由新到舊排序，名稱唯一（Trim + 忽略大小寫，編輯排除自身）。

Tech Stack: .NET 10、Blazor Server、Ant Design Blazor、EF Core（Sqlite + SQL Server 雙 provider）、AutoMapper、xUnit。

using AutoMapper;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;
using KnowledgeExtraction.AccessDatas;
using KnowledgeExtraction.AccessDatas.Models;
using KnowledgeExtraction.Business.Factories;
using KnowledgeExtraction.Business.Helpers;
using KnowledgeExtraction.Models.AdapterModel;
using KnowledgeExtraction.Models.Systems;

namespace KnowledgeExtraction.Business.Services.DataAccess;

public class PromptService
{
    private readonly BackendDBContext context;

    public IMapper Mapper { get; }
    public ILogger<PromptService> Logger { get; }

    public PromptService(
        BackendDBContext context,
        IMapper mapper,
        ILogger<PromptService> logger)
    {
        this.context = context;
        Mapper = mapper;
        Logger = logger;
    }

    public async Task<DataRequestResult<PromptAdapterModel>> GetAsync(DataRequest dataRequest)
    {
        DataRequestResult<PromptAdapterModel> result = new();
        IQueryable<Prompt> dataSource = context.Prompt.AsNoTracking();

        if (!string.IsNullOrWhiteSpace(dataRequest.Search))
        {
            var search = dataRequest.Search.Trim();
            dataSource = dataSource.Where(x =>
                x.Name.Contains(search) ||
                (x.Description ?? string.Empty).Contains(search));
        }

        #region 排序 (預設為更新時間新到舊)
        if (dataRequest.SortField == nameof(PromptAdapterModel.Name))
        {
            dataSource = dataRequest.SortDescending == true
                ? dataSource.OrderByDescending(x => x.Name).ThenByDescending(x => x.Id)
                : dataRequest.SortDescending == false
                    ? dataSource.OrderBy(x => x.Name).ThenBy(x => x.Id)
                    : dataSource.OrderByDescending(x => x.UpdatedAt).ThenByDescending(x => x.Id);
        }
        else if (dataRequest.SortField == nameof(PromptAdapterModel.IsEnabled))
        {
            dataSource = dataRequest.SortDescending == true
                ? dataSource.OrderByDescending(x => x.IsEnabled).ThenByDescending(x => x.Id)
                : dataRequest.SortDescending == false
                    ? dataSource.OrderBy(x => x.IsEnabled).ThenBy(x => x.Id)
                    : dataSource.OrderByDescending(x => x.UpdatedAt).ThenByDescending(x => x.Id);
        }
        else if (dataRequest.SortField == nameof(PromptAdapterModel.CreatedAt))
        {
            dataSource = dataRequest.SortDescending == true
                ? dataSource.OrderByDescending(x => x.CreatedAt).ThenByDescending(x => x.Id)
                : dataRequest.SortDescending == false
                    ? dataSource.OrderBy(x => x.CreatedAt).ThenBy(x => x.Id)
                    : dataSource.OrderByDescending(x => x.UpdatedAt).ThenByDescending(x => x.Id);
        }
        else if (dataRequest.SortField == nameof(PromptAdapterModel.UpdatedAt))
        {
            dataSource = dataRequest.SortDescending == false
                ? dataSource.OrderBy(x => x.UpdatedAt).ThenBy(x => x.Id)
                : dataSource.OrderByDescending(x => x.UpdatedAt).ThenByDescending(x => x.Id);
        }
        else
        {
            // 預設：最近更新時間新到舊
            dataSource = dataSource.OrderByDescending(x => x.UpdatedAt).ThenByDescending(x => x.Id);
        }
        #endregion

        result.Count = await dataSource.CountAsync();
        dataSource = dataSource.Skip((dataRequest.CurrentPage - 1) * dataRequest.PageSize);
        if (dataRequest.Take != 0)
        {
            dataSource = dataSource.Take(dataRequest.PageSize);
        }

        var records = await dataSource.ToListAsync();
        result.Result = Mapper.Map<List<PromptAdapterModel>>(records);
        return result;
    }

    public async Task<PromptAdapterModel> GetAsync(int id)
    {
        Prompt? item = await context.Prompt
            .AsNoTracking()
            .FirstOrDefaultAsync(x => x.Id == id);

        if (item is null)
        {
            Logger.LogWarning("Prompt not found. PromptId={PromptId}", id);
            return new PromptAdapterModel();
        }

        return Mapper.Map<PromptAdapterModel>(item);
    }

    public async Task<VerifyRecordResult> AddAsync(PromptAdapterModel paraObject)
    {
        try
        {
            CleanTrackingHelper.Clean<Prompt>(context);
            Prompt itemParameter = Mapper.Map<Prompt>(paraObject);

            await context.Prompt.AddAsync(itemParameter);
            await context.SaveChangesAsync();
            CleanTrackingHelper.Clean<Prompt>(context);

            return VerifyRecordResultFactory.Build(true);
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Failed to create prompt. Name={Name}", paraObject.Name);
            return VerifyRecordResultFactory.Build(false, "新增提示詞失敗。", ex);
        }
    }

    public async Task<VerifyRecordResult> UpdateAsync(PromptAdapterModel paraObject)
    {
        try
        {
            CleanTrackingHelper.Clean<Prompt>(context);
            Prompt? currentItem = await context.Prompt
                .FirstOrDefaultAsync(x => x.Id == paraObject.Id);

            if (currentItem == null)
            {
                return VerifyRecordResultFactory.Build(false, "找不到要修改的提示詞資料。");
            }

            currentItem.Name = paraObject.Name;
            currentItem.Content = paraObject.Content;
            currentItem.Description = paraObject.Description;
            currentItem.IsEnabled = paraObject.IsEnabled;
            currentItem.UpdatedAt = DateTime.Now;

            await context.SaveChangesAsync();
            CleanTrackingHelper.Clean<Prompt>(context);

            return VerifyRecordResultFactory.Build(true);
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Failed to update prompt. PromptId={PromptId}, Name={Name}", paraObject.Id, paraObject.Name);
            return VerifyRecordResultFactory.Build(false, "修改提示詞失敗。", ex);
        }
    }

    public async Task<VerifyRecordResult> DeleteAsync(int id)
    {
        try
        {
            CleanTrackingHelper.Clean<Prompt>(context);
            Prompt? item = await context.Prompt.FirstOrDefaultAsync(x => x.Id == id);

            if (item == null)
            {
                return VerifyRecordResultFactory.Build(false, "找不到要刪除的提示詞資料。");
            }

            context.Prompt.Remove(item);
            await context.SaveChangesAsync();
            CleanTrackingHelper.Clean<Prompt>(context);

            return VerifyRecordResultFactory.Build(true);
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Failed to delete prompt. PromptId={PromptId}", id);
            return VerifyRecordResultFactory.Build(false, "刪除提示詞失敗。", ex);
        }
    }

    public async Task<VerifyRecordResult> BeforeAddCheckAsync(PromptAdapterModel paraObject)
    {
        return await ValidateNameUniqueAsync(paraObject);
    }

    public async Task<VerifyRecordResult> BeforeUpdateCheckAsync(PromptAdapterModel paraObject)
    {
        CleanTrackingHelper.Clean<Prompt>(context);
        Prompt? searchItem = await context.Prompt
            .AsNoTracking()
            .FirstOrDefaultAsync(x => x.Id == paraObject.Id);

        if (searchItem == null)
        {
            return VerifyRecordResultFactory.Build(false, "要修改的提示詞資料不存在。");
        }

        return await ValidateNameUniqueAsync(paraObject, paraObject.Id);
    }

    public Task<VerifyRecordResult> BeforeDeleteCheckAsync(PromptAdapterModel paraObject)
    {
        return Task.FromResult(VerifyRecordResultFactory.Build(true));
    }

    /// <summary>
    /// 驗證提示詞名稱唯一性 (忽略大小寫並去除頭尾空白)
    /// </summary>
    private async Task<VerifyRecordResult> ValidateNameUniqueAsync(PromptAdapterModel paraObject, int? excludeId = null)
    {
        var normalized = (paraObject.Name ?? string.Empty).Trim().ToLower();

        if (string.IsNullOrWhiteSpace(normalized))
        {
            return VerifyRecordResultFactory.Build(false, "提示詞名稱 不可為空白。");
        }

        var query = context.Prompt
            .AsNoTracking()
            .Where(x => x.Name.Trim().ToLower() == normalized);

        if (excludeId.HasValue)
        {
            query = query.Where(x => x.Id != excludeId.Value);
        }

        if (await query.AnyAsync())
        {
            return VerifyRecordResultFactory.Build(false, $"提示詞名稱 '{(paraObject.Name ?? string.Empty).Trim()}' 已存在。");
        }

        return VerifyRecordResultFactory.Build(true);
    }
}

    public async Task<PagedResult<Prompt>> GetPagedAsync(PromptSearchRequestDto request)
    {
        var query = context.Prompt.AsNoTracking().AsQueryable();

        if (!string.IsNullOrWhiteSpace(request.Keyword))
        {
            var keyword = request.Keyword.Trim();
            query = query.Where(c =>
                c.Name.Contains(keyword) ||
                (c.Description != null && c.Description.Contains(keyword)));
        }

        if (request.IsEnabled.HasValue)
        {
            query = query.Where(c => c.IsEnabled == request.IsEnabled.Value);
        }

        #region 根據 SortBy 及 SortDescending 排序 (預設為更新時間新到舊)
        query = request.SortBy?.ToLower() switch
        {
            "name" => request.SortDescending
                ? query.OrderByDescending(c => c.Name)
                : query.OrderBy(c => c.Name),
            "isenabled" => request.SortDescending
                ? query.OrderByDescending(c => c.IsEnabled)
                : query.OrderBy(c => c.IsEnabled),
            "createdat" => request.SortDescending
                ? query.OrderByDescending(c => c.CreatedAt)
                : query.OrderBy(c => c.CreatedAt),
            "updatedat" => request.SortDescending
                ? query.OrderByDescending(c => c.UpdatedAt)
                : query.OrderBy(c => c.UpdatedAt),
            _ => query.OrderByDescending(c => c.UpdatedAt),
        };
        #endregion

        var totalCount = await query.CountAsync();

        var items = await query
            .Skip((request.PageIndex - 1) * request.PageSize)
            .Take(request.PageSize)
            .ToListAsync();

        return new PagedResult<Prompt>
        {
            Items = items,
            PageIndex = request.PageIndex,
            PageSize = request.PageSize,
            TotalCount = totalCount
        };
    }

    /// <summary>
    /// 檢查提示詞名稱是否存在 (忽略大小寫並去除頭尾空白)
    /// </summary>
    public async Task<bool> ExistsByNameAsync(string name, int? excludeId = null)
    {
        var normalized = (name ?? string.Empty).Trim().ToLower();

        var query = context.Prompt.Where(c => c.Name.Trim().ToLower() == normalized);

        if (excludeId.HasValue)
        {
            query = query.Where(c => c.Id != excludeId.Value);
        }

        return await query.AnyAsync();
    }

    #endregion

    #region 新增方法

    public async Task<Prompt> AddAsync(Prompt prompt)
    {
        prompt.CreatedAt = DateTime.Now;
        prompt.UpdatedAt = DateTime.Now;

        await context.Prompt.AddAsync(prompt);
        await context.SaveChangesAsync();

        return prompt;
    }

    #endregion

    #region 更新方法

    public async Task<bool> UpdateAsync(Prompt prompt)
    {
        var existingPrompt = await context.Prompt.FindAsync(prompt.Id);
        if (existingPrompt == null)
        {
            return false;
        }

        prompt.UpdatedAt = DateTime.Now;
        prompt.CreatedAt = existingPrompt.CreatedAt; // 保留原建立時間

        context.Entry(existingPrompt).CurrentValues.SetValues(prompt);
        await context.SaveChangesAsync();

        return true;
    }

    #endregion

    #region 刪除方法

    public async Task<bool> DeleteAsync(int id)
    {
        var prompt = await context.Prompt.FindAsync(id);
        if (prompt == null)
        {
            return false;
        }

        context.Prompt.Remove(prompt);
        await context.SaveChangesAsync();

        return true;
    }

        string? columnFieldName = column.GetType().GetProperty("FieldName")?.GetValue(column)?.ToString();
        if (!string.IsNullOrWhiteSpace(columnFieldName))
        {
            return columnFieldName;
        }

        object? dataIndex = column.GetType().GetProperty("DataIndex")?.GetValue(column);
        return dataIndex?.ToString() ?? string.Empty;
    }

    private async Task OnSearchAsync()
    {
        _pageIndex = 1;
        await ReloadAsync();
    }

    private async Task OnRefreshAsync()
    {
        await ReloadAsync();

        _ = notificationService.Open(new NotificationConfig
        {
            Message = "系統訊息",
            Description = "已更新最新資料",
            NotificationType = NotificationType.Warning,
            Placement = NotificationPlacement.BottomRight
        });
    }

    private async Task OnEditAsync(PromptAdapterModel promptAdapterModel)
    {
        isNewRecordMode = false;
        modalTitle = "修改提示詞";
        CurrentRecord = await promptService.GetAsync(promptAdapterModel.Id);
        modalVisible = true;
    }

    private async Task OnDeleteAsync(PromptAdapterModel promptAdapterModel)
    {
        var beforeDeleteCheckResult = await promptService.BeforeDeleteCheckAsync(promptAdapterModel);
        if (!beforeDeleteCheckResult.Success)
        {
            _ = notificationService.Open(new NotificationConfig
            {
                Message = "系統訊息",
                Description = beforeDeleteCheckResult.Message,
                NotificationType = NotificationType.Error,
                Placement = NotificationPlacement.BottomRight
            });
            return;
        }

        var ok = await modalService.ConfirmAsync(new ConfirmOptions
        {
            Title = "確認刪除",
            Content = "確定要刪除這筆紀錄嗎？此操作無法復原。",
            OkText = "刪除",
            CancelText = "取消",
            OkButtonProps = new ButtonProps { Danger = true },
            MaskClosable = false
        });

        if (!ok)
        {
            return;
        }

        await promptService.DeleteAsync(promptAdapterModel.Id);

        _ = notificationService.Open(new NotificationConfig
        {
            Message = "系統訊息",
            Description = "刪除成功",
            NotificationType = NotificationType.Warning,
            Placement = NotificationPlacement.BottomRight
        });

        await ReloadAsync();
    }

    private Task OnAddAsync()
    {
        CurrentRecord = new PromptAdapterModel
        {
            IsEnabled = true
        };

        isNewRecordMode = true;
        modalTitle = "新增提示詞";
        modalVisible = true;
        return Task.CompletedTask;
    }

    private async Task OnModalOKHandleAsync(MouseEventArgs args)
    {
        if (LocalEditContext?.Validate() == false)
        {
            IEnumerable<string> allErrors = LocalEditContext.GetValidationMessages();
            foreach (var error in allErrors)
            {
                _ = notificationService.Open(new NotificationConfig
                {
                    Message = "驗證失敗",
                    Description = error,
                    NotificationType = NotificationType.Error,
                    Placement = NotificationPlacement.BottomRight,
                    Duration = 5
                });
            }

            modalVisible = true;
            return;
        }

        VerifyRecordResult actionResult;

        if (isNewRecordMode)
        {
            var beforeAddCheckResult = await promptService.BeforeAddCheckAsync(CurrentRecord);
            if (!beforeAddCheckResult.Success)
            {
                _ = notificationService.Open(new NotificationConfig
                {
                    Message = "系統訊息",
                    Description = beforeAddCheckResult.Message,
                    NotificationType = NotificationType.Error,
                    Placement = NotificationPlacement.BottomRight
                });

                modalVisible = true;
                return;
            }

            CurrentRecord.CreatedAt = DateTime.Now;
            CurrentRecord.UpdatedAt = DateTime.Now;

            actionResult = await promptService.AddAsync(CurrentRecord);
        }
        else
        {
            var beforeUpdateCheckResult = await promptService.BeforeUpdateCheckAsync(CurrentRecord);
            if (!beforeUpdateCheckResult.Success)
            {
                _ = notificationService.Open(new NotificationConfig
                {
                    Message = "系統訊息",
                    Description = beforeUpdateCheckResult.Message,
                    NotificationType = NotificationType.Error,
                    Placement = NotificationPlacement.BottomRight
                });

                modalVisible = true;
                return;
            }

            CurrentRecord.UpdatedAt = DateTime.Now;
            actionResult = await promptService.UpdateAsync(CurrentRecord);
        }

        if (!actionResult.Success)
        {
            _ = notificationService.Open(new NotificationConfig
            {
                Message = "系統訊息",
                Description = actionResult.Message,
                NotificationType = NotificationType.Error,
                Placement = NotificationPlacement.BottomRight
            });

            modalVisible = true;
            return;
        }

        _ = notificationService.Open(new NotificationConfig
        {
            Message = "系統訊息",
            Description = isNewRecordMode ? "新增成功" : "修改成功",
            NotificationType = NotificationType.Warning,
            Placement = NotificationPlacement.BottomRight
        });

        if (isNewRecordMode)
        {
            _ = messageService.SuccessAsync("新增成功");
        }

        await ReloadAsync();
        modalVisible = false;
    }

    private Task OnModalCancelHandleAsync(MouseEventArgs args)
    {
        modalVisible = false;
        return Task.CompletedTask;
    }

    private async Task OnModalKeyDownAsync(KeyboardEventArgs args)
    {
        if (args.Key == "Escape" || args.Key == "Esc")
        {
            await OnModalCancelHandleAsync(new MouseEventArgs());
        }
    }

    public void OnEditContestChanged(EditContext context)
    {
        LocalEditContext = context;
    }
}
