知識來源管理清單（KnowledgeSource）CRUD + 成員聚合 實作計畫

Archived historical implementation plan: 本檔保留當時 agent 執行計畫與脈絡；下列步驟不再作為目前 backlog 追蹤。

Goal: 新增「知識來源管理清單」頁面，提供知識來源的 CRUD 與跨三管線（音訊轉錄／文件萃取／圖片清單）的成員聚合管理，並於側邊欄新增父選單「清洗與蒸餾」與角色權限。

Architecture: 比照 DocumentExtraction 的分層（Entity → AdapterModel → Service → Blazor View），但不上傳檔案、不接佇列；新增 KnowledgeSource（本體）與 KnowledgeSourceItem（成員，軟參照 ItemType+ItemId，Cascade Delete）。團隊權控沿用 IRecordAccessScopeProvider + TagStringHelper。清單預設依 UpdatedAt 由新到舊。選單／權限三處以位置索引同步。

Tech Stack: .NET 10、Blazor Server、Ant Design Blazor、EF Core（SQLite + SQL Server 雙 provider）、AutoMapper、xUnit（SQLite in-memory 測試）。

規格來源： docs/prd/知識來源管理-prd.md

namespace KnowledgeExtraction.Business.Services.DataAccess;

public class KnowledgeSourceService
{
    private const string ItemStatusCompleted = "已完成";

    private readonly BackendDBContext context;
    private readonly IRecordAccessScopeProvider accessScope;

    public IMapper Mapper { get; }
    public ILogger<KnowledgeSourceService> Logger { get; }

    public KnowledgeSourceService(
        BackendDBContext context,
        IMapper mapper,
        ILogger<KnowledgeSourceService> logger,
        IRecordAccessScopeProvider accessScope)
    {
        this.context = context;
        Mapper = mapper;
        Logger = logger;
        this.accessScope = accessScope;
    }

    public async Task<DataRequestResult<KnowledgeSourceAdapterModel>> GetAsync(DataRequest dataRequest)
    {
        DataRequestResult<KnowledgeSourceAdapterModel> result = new();
        IQueryable<KnowledgeSource> dataSource = context.KnowledgeSource.AsNoTracking();

        if (!string.IsNullOrWhiteSpace(dataRequest.Search))
        {
            var search = dataRequest.Search.Trim();
            dataSource = dataSource.Where(x =>
                x.Title.Contains(search) ||
                (x.Description ?? string.Empty).Contains(search));
        }

        if (dataRequest.CategoryFilters.Count > 0)
        {
            dataSource = dataSource.Where(TagStringHelper.BuildContainsAnyPredicate<KnowledgeSource>(x => x.Categories, dataRequest.CategoryFilters));
        }

        if (dataRequest.TeamFilters.Count > 0)
        {
            dataSource = dataSource.Where(TagStringHelper.BuildContainsAnyPredicate<KnowledgeSource>(x => x.Teams, dataRequest.TeamFilters));
        }

        var scope = await accessScope.GetAsync();
        if (!scope.IsAdmin)
        {
            dataSource = dataSource.Where(TagStringHelper.BuildTeamAccessPredicate<KnowledgeSource>(x => x.Teams, scope.Teams));
        }

        dataSource = (dataRequest.SortField, dataRequest.SortDescending) switch
        {
            (nameof(KnowledgeSourceAdapterModel.Title), true) => dataSource.OrderByDescending(x => x.Title).ThenByDescending(x => x.Id),
            (nameof(KnowledgeSourceAdapterModel.Title), false) => dataSource.OrderBy(x => x.Title).ThenBy(x => x.Id),
            (nameof(KnowledgeSourceAdapterModel.CreatedAt), true) => dataSource.OrderByDescending(x => x.CreatedAt).ThenByDescending(x => x.Id),
            (nameof(KnowledgeSourceAdapterModel.CreatedAt), false) => dataSource.OrderBy(x => x.CreatedAt).ThenBy(x => x.Id),
            (nameof(KnowledgeSourceAdapterModel.UpdatedAt), false) => dataSource.OrderBy(x => x.UpdatedAt).ThenBy(x => x.Id),
            // 預設與 UpdatedAt 降序：最近更新在最前
            _ => dataSource.OrderByDescending(x => x.UpdatedAt).ThenByDescending(x => x.Id)
        };

        result.Count = await dataSource.CountAsync();
        dataSource = dataSource.Skip((dataRequest.CurrentPage - 1) * dataRequest.PageSize);
        if (dataRequest.Take != 0)
        {
            dataSource = dataSource.Take(dataRequest.PageSize);
        }

        var records = await dataSource
            .Select(x => new { Entity = x, ItemCount = x.Items.Count })
            .ToListAsync();

        var models = new List<KnowledgeSourceAdapterModel>();
        foreach (var row in records)
        {
            var model = Mapper.Map<KnowledgeSourceAdapterModel>(row.Entity);
            model.ItemCount = row.ItemCount;
            models.Add(model);
        }
        result.Result = models;
        return result;
    }

    public async Task<KnowledgeSourceAdapterModel> GetAsync(int id)
    {
        var item = await context.KnowledgeSource.AsNoTracking()
            .Include(x => x.Items)
            .FirstOrDefaultAsync(x => x.Id == id);
        if (item is null)
        {
            return new KnowledgeSourceAdapterModel();
        }

        var scope = await accessScope.GetAsync();
        if (!TagStringHelper.IsTeamAccessible(item.Teams, scope.Teams, scope.IsAdmin))
        {
            Logger.LogWarning("Knowledge source access denied by team scope. Id={Id}", id);
            return new KnowledgeSourceAdapterModel();
        }

        var model = Mapper.Map<KnowledgeSourceAdapterModel>(item);
        model.ItemCount = item.Items.Count;
        model.Items = new List<KnowledgeSourceItemAdapterModel>();
        foreach (var member in item.Items.OrderBy(x => x.SortOrder).ThenBy(x => x.Id))
        {
            var memberModel = Mapper.Map<KnowledgeSourceItemAdapterModel>(member);
            memberModel.ItemTypeName = KnowledgeSourceItemAdapterModel.ResolveTypeName(member.ItemType);
            memberModel.ItemTitle = await ResolveItemTitleAsync(member.ItemType, member.ItemId);
            model.Items.Add(memberModel);
        }
        return model;
    }

    public async Task<VerifyRecordResult> AddAsync(KnowledgeSourceAdapterModel paraObject)
    {
        try
        {
            var now = DateTime.Now;
            var entity = new KnowledgeSource
            {
                Title = paraObject.Title,
                Description = paraObject.Description,
                Categories = TagStringHelper.ToStored(paraObject.Categories),
                Teams = TagStringHelper.ToStored(paraObject.Teams),
                CreatedAt = now,
                UpdatedAt = now
            };
            await context.KnowledgeSource.AddAsync(entity);
            await context.SaveChangesAsync();
            return VerifyRecordResultFactory.Build(true);
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Failed to create knowledge source. Title={Title}", paraObject.Title);
            return VerifyRecordResultFactory.Build(false, "新增知識來源失敗。", ex);
        }
    }

    public async Task<VerifyRecordResult> UpdateAsync(KnowledgeSourceAdapterModel paraObject)
    {
        try
        {
            var current = await context.KnowledgeSource.FirstOrDefaultAsync(x => x.Id == paraObject.Id);
            if (current is null)
            {
                return VerifyRecordResultFactory.Build(false, "找不到要修改的知識來源。");
            }

            var scope = await accessScope.GetAsync();
            if (!TagStringHelper.IsTeamAccessible(current.Teams, scope.Teams, scope.IsAdmin))
            {
                return VerifyRecordResultFactory.Build(false, "您沒有此紀錄的存取權限。");

    public async Task<VerifyRecordResult> MoveItemAsync(int itemId, bool up)
    {
        try
        {
            var member = await context.KnowledgeSourceItem.Include(x => x.KnowledgeSource).FirstOrDefaultAsync(x => x.Id == itemId);
            if (member is null || member.KnowledgeSource is null)

            {
                return VerifyRecordResultFactory.Build(false, "您沒有此知識來源的存取權限。");
            }

            var siblings = await context.KnowledgeSourceItem
                .Where(x => x.KnowledgeSourceId == member.KnowledgeSourceId)
                .OrderBy(x => x.SortOrder).ThenBy(x => x.Id)
                .ToListAsync();
            var index = siblings.FindIndex(x => x.Id == itemId);
            var swapIndex = up ? index - 1 : index + 1;
            if (index < 0 || swapIndex < 0 || swapIndex >= siblings.Count)
            {
                return VerifyRecordResultFactory.Build(true); // 已在邊界，無需移動
            }

            (siblings[index].SortOrder, siblings[swapIndex].SortOrder) = (siblings[swapIndex].SortOrder, siblings[index].SortOrder);
            member.KnowledgeSource.UpdatedAt = DateTime.Now;
            await context.SaveChangesAsync();
            return VerifyRecordResultFactory.Build(true);
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "Failed to move knowledge source item. ItemId={ItemId}", itemId);
            return VerifyRecordResultFactory.Build(false, "調整成員順序失敗。", ex);
        }
    }

        var r = await service.AddItemAsync(memberSource.Id, item.ItemType, item.ItemId);
        if (!r.Success) { ShowError(r.Message); return; }
        memberSource = await service.GetAsync(memberSource.Id);
        await ReloadSelectableAsync();
    }

    private async Task OnRemoveMemberAsync(KnowledgeSourceItemAdapterModel member)
    {
        var r = await service.RemoveItemAsync(member.Id);
        if (!r.Success) { ShowError(r.Message); return; }
        memberSource = await service.GetAsync(memberSource.Id);
        await ReloadSelectableAsync();
    }

    private async Task OnMoveMemberAsync(KnowledgeSourceItemAdapterModel member, bool up)
    {
        var r = await service.MoveItemAsync(member.Id, up);
        if (!r.Success) { ShowError(r.Message); return; }
        memberSource = await service.GetAsync(memberSource.Id);
    }

    private async Task OnMemberModalCloseAsync()
    {
        memberModalVisible = false;
        await ReloadAsync(); // 成員數/更新時間可能變動
    }

    private void ShowError(string message)
    {
        _ = notificationService.Open(new NotificationConfig
        {
            Message = "系統訊息",
            Description = message,
            NotificationType = NotificationType.Error,
            Placement = NotificationPlacement.BottomRight
        });
    }
