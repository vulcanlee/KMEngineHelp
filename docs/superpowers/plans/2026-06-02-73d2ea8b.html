音訊／影片轉錄服務 Implementation Plan

Archived historical implementation plan: 本檔保留當時 agent 執行計畫與脈絡；下列步驟不再作為目前 backlog 追蹤。

Goal: 新增一個 CRUD 頁面，讓使用者上傳音訊／影片檔案，並透過背景工作呼叫 OpenAI gpt-4o-transcribe 產生繁體中文逐字稿（依年度切分目錄存成文字檔），前端可即時看到轉換進度。

Architecture: 沿用既有 ProjectPage 的 Blazor Server + AntDesign + EF Core 分層（Page → View → Service → DbContext / Repository → Entity，DTO + AutoMapper）。新增一條輕量背景管線：Channel<int> 工作佇列（Singleton）+ BackgroundService 消費者，消費者開 scope 取得 TranscriptionProcessor（呼叫 OpenAI、必要時用 FFmpeg 分段），全程更新資料庫狀態欄位；前端以既有「重新整理」流程輪詢狀態。

Tech Stack: .NET 10 / ASP.NET Core Blazor Server、AntDesign 1.6、EF Core 10（SQLite + SqlServer 雙 migration assembly）、AutoMapper、System.Threading.Channels、IHttpClientFactory、外部 FFmpeg、xUnit。

PRD 來源： docs/prd/音訊影片轉錄服務PRD.md

重要前提與既有慣例

  • 權限種子自動化： Program.cs:297 每次啟動都會把 RolePermissionService.GetRolePermissionAllNameToJson() 的全部權限寫回預設角色的 TabViewJson。因此只要在 RolePermissionService 與 MagicObjectHelper 加入新權限字串，下次啟動預設角色即自動擁有，不需改資料。
  • 目錄自動建立： Program.cs:234-239 用 EnsureDirectoryExists(...) 對每個設定路徑建目錄；新路徑要照加。
  • 既有檔案下載端點未實作： UI 參照的 /api/project-files/{id}/download 等路由在程式中並無對應 controller（既有未完成連結）。本功能改為實作真正可用的逐字稿檢視／下載端點，不沿用該斷鏈。
  • 資料模型簡化（YAGNI）： 一筆轉錄紀錄對應「一個媒體檔 + 一份逐字稿」，因此採單一資料表 MediaTranscription（媒體與逐字稿皆以相對路徑欄位指向磁碟檔），不另建 *File 子表。
  • 版本號： 完成建置時，依 CLAUDE.md 將 appsettings.json 的 SystemSettings.SystemInformation.SystemVersion +1（見最後 Task）。

namespace KnowledgeExtraction.Business.Helpers;

public static class MediaFormatValidator
{
    // OpenAI 語音轉文字支援之副檔名
    public static readonly IReadOnlySet<string> AllowedExtensions = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
    {
        ".flac", ".m4a", ".mp3", ".mp4", ".mpeg", ".mpga", ".oga", ".ogg", ".wav", ".webm"
    };

    public static bool IsSupported(string? fileName)
    {
        if (string.IsNullOrWhiteSpace(fileName))
        {
            return false;
        }

        var ext = Path.GetExtension(fileName);
        return !string.IsNullOrEmpty(ext) && AllowedExtensions.Contains(ext);
    }

    public static string AllowedExtensionsText => string.Join("、", AllowedExtensions);
}

using System.Diagnostics;

namespace KnowledgeExtraction.Business.Services.Transcription;

public class MediaChunker
{
    private readonly string ffmpegPath;

    public MediaChunker(string ffmpegPath)
    {
        this.ffmpegPath = string.IsNullOrWhiteSpace(ffmpegPath) ? "ffmpeg" : ffmpegPath;
    }

    public static bool NeedsChunking(long fileSizeBytes, int maxChunkSizeMB)
        => fileSizeBytes > (long)maxChunkSizeMB * 1024 * 1024;

    public static int EstimateChunkCount(long fileSizeBytes, int maxChunkSizeMB)
    {
        var max = (long)maxChunkSizeMB * 1024 * 1024;
        if (max <= 0)
        {
            return 1;
        }
        return (int)Math.Max(1, Math.Ceiling(fileSizeBytes / (double)max));
    }

    /// <summary>
    /// 以位元率估算每段秒數，使用 ffmpeg segment 將輸入切成多個音檔片段，回傳片段完整路徑（依序）。
    /// </summary>
    public async Task<List<string>> SplitAsync(
        string inputFullPath,
        string outputDirectory,
        int chunkCount,
        CancellationToken cancellationToken)
    {
        Directory.CreateDirectory(outputDirectory);

        var totalSeconds = await GetDurationSecondsAsync(inputFullPath, cancellationToken);
        var segmentSeconds = Math.Max(1, (int)Math.Ceiling(totalSeconds / chunkCount));
        var pattern = Path.Combine(outputDirectory, "chunk_%03d.mp3");

        // 統一輸出為 mp3，確保切割後每段為 Whisper 支援格式且體積可控
        var args = $"-y -i \"{inputFullPath}\" -f segment -segment_time {segmentSeconds} -vn -acodec libmp3lame -q:a 4 \"{pattern}\"";
        await RunFfmpegAsync(args, cancellationToken);

        return Directory.GetFiles(outputDirectory, "chunk_*.mp3").OrderBy(x => x).ToList();
    }

    private async Task<double> GetDurationSecondsAsync(string inputFullPath, CancellationToken cancellationToken)
    {
        // 以 ffmpeg 讀取 stderr 解析時長；失敗則回傳保守值
        var psi = new ProcessStartInfo
        {
            FileName = ffmpegPath,
            Arguments = $"-i \"{inputFullPath}\"",
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true
        };
        using var process = Process.Start(psi)!;
        var stderr = await process.StandardError.ReadToEndAsync(cancellationToken);
        await process.WaitForExitAsync(cancellationToken);

        var match = System.Text.RegularExpressions.Regex.Match(stderr, @"Duration: (\d+):(\d+):(\d+\.\d+)");
        if (match.Success)
        {
            return int.Parse(match.Groups[1].Value) * 3600
                 + int.Parse(match.Groups[2].Value) * 60
                 + double.Parse(match.Groups[3].Value);
        }
        return 0;
    }

    private async Task RunFfmpegAsync(string arguments, CancellationToken cancellationToken)
    {
        var psi = new ProcessStartInfo
        {
            FileName = ffmpegPath,
            Arguments = arguments,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true
        };
        using var process = Process.Start(psi)
            ?? throw new InvalidOperationException("無法啟動 FFmpeg，請確認已安裝並設定 OpenAISettings:FfmpegPath。");
        var stderr = await process.StandardError.ReadToEndAsync(cancellationToken);
        await process.WaitForExitAsync(cancellationToken);
        if (process.ExitCode != 0)
        {
            throw new InvalidOperationException($"FFmpeg 執行失敗 (ExitCode={process.ExitCode})：{stderr}");
        }
    }

public class TranscriptionProcessor

    private readonly OpenAISettings settings;
    private readonly string mediaRootPath;
    private readonly string transcriptRootPath;
    private readonly ILogger<TranscriptionProcessor> logger;

    public TranscriptionProcessor(
        BackendDBContext context,
        OpenAITranscriptionClient client,
        IOptions<OpenAISettings> openAiOptions,
        IOptions<SystemSettings> systemSettings,
        ILogger<TranscriptionProcessor> logger)
    {
        this.context = context;
        this.client = client;
        settings = openAiOptions.Value;
        mediaRootPath = systemSettings.Value.ExternalFileSystem.MediaUploadPath;
        transcriptRootPath = systemSettings.Value.ExternalFileSystem.TranscriptionFilePath;
        this.logger = logger;

    public async Task ProcessAsync(int id, CancellationToken cancellationToken)
    {
        var record = await context.MediaTranscription.FirstOrDefaultAsync(x => x.Id == id, cancellationToken);
        if (record is null)
        {
            logger.LogWarning("Transcription record not found. Id={Id}", id);
            return;
        }

        var tempDir = Path.Combine(Path.GetTempPath(), $"transcribe_{id}_{Guid.NewGuid():N}");

        try
        {
            await UpdateStatusAsync(record, MediaTranscriptionAdapterModel.StatusProcessing, 0, null, cancellationToken);

            var mediaFullPath = Path.Combine(mediaRootPath, NormalizePath(record.MediaRelativePath));
            var chunker = new MediaChunker(settings.FfmpegPath);

            List<string> partsToTranscribe;
            if (MediaChunker.NeedsChunking(record.FileSize, settings.MaxChunkSizeMB))
            {
                var chunkCount = MediaChunker.EstimateChunkCount(record.FileSize, settings.MaxChunkSizeMB);
                logger.LogInformation("Splitting media into {Count} chunks. Id={Id}", chunkCount, id);
                partsToTranscribe = await chunker.SplitAsync(mediaFullPath, tempDir, chunkCount, cancellationToken);
            }
            else
            {
                partsToTranscribe = [mediaFullPath];
            }

            var segments = new List<string>();
            for (var i = 0; i < partsToTranscribe.Count; i++)
            {
                cancellationToken.ThrowIfCancellationRequested();
                var text = await client.TranscribeAsync(partsToTranscribe[i], cancellationToken);
                segments.Add(text);

                var progress = (int)Math.Round((i + 1) / (double)partsToTranscribe.Count * 100);
                await UpdateStatusAsync(record, MediaTranscriptionAdapterModel.StatusProcessing, progress, null, cancellationToken);
            }

            var transcript = string.Join(Environment.NewLine, segments).Trim();
            var transcriptRelativePath = await SaveTranscriptAsync(record, transcript, cancellationToken);

            record.TranscriptRelativePath = transcriptRelativePath;
            record.Model = settings.Model;
            record.Language = settings.Language;
            record.CompletedAt = DateTime.Now;
            await UpdateStatusAsync(record, MediaTranscriptionAdapterModel.StatusCompleted, 100, null, cancellationToken);
            logger.LogInformation("Transcription completed. Id={Id}", id);
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Transcription failed. Id={Id}", id);
            await UpdateStatusAsync(record, MediaTranscriptionAdapterModel.StatusFailed, record.ProgressPercentage, ex.Message, cancellationToken);
        }
        finally
        {
            TryDeleteDirectory(tempDir);
        }
    }

    private async Task<string> SaveTranscriptAsync(MediaTranscription record, string transcript, CancellationToken cancellationToken)
    {
        var year = record.UploadedAt.Year.ToString("0000");
        var relativePath = Path.Combine(year, $"{Guid.NewGuid():N}.txt");
        var fullPath = Path.Combine(transcriptRootPath, relativePath);
        var directory = Path.GetDirectoryName(fullPath);
        if (!string.IsNullOrWhiteSpace(directory))
        {
            Directory.CreateDirectory(directory);
        }

        await File.WriteAllTextAsync(fullPath, transcript, new System.Text.UTF8Encoding(false), cancellationToken);
        return relativePath.Replace('\\', '/');
    }

    private async Task UpdateStatusAsync(MediaTranscription record, string status, int progress, string? error, CancellationToken cancellationToken)
    {
        record.Status = status;
        record.ProgressPercentage = progress;
        record.ErrorMessage = error;
        record.UpdatedAt = DateTime.Now;
        await context.SaveChangesAsync(cancellationToken);
    }

    private static string NormalizePath(string relativePath)
        => relativePath.Replace('/', Path.DirectorySeparatorChar).Replace('\\', Path.DirectorySeparatorChar);

    private void TryDeleteDirectory(string dir)
    {
        try
        {
            if (Directory.Exists(dir))
            {
                Directory.Delete(dir, recursive: true);
            }
        }
        catch (Exception ex)
        {
            logger.LogWarning(ex, "Failed to clean temp dir {Dir}", dir);
        }
    }
}

    public async Task<DataRequestResult<MediaTranscriptionAdapterModel>> GetAsync(DataRequest dataRequest)
    {
        DataRequestResult<MediaTranscriptionAdapterModel> result = new();
        IQueryable<MediaTranscription> dataSource = context.MediaTranscription.AsNoTracking();

        if (!string.IsNullOrWhiteSpace(dataRequest.Search))

                (nameof(MediaTranscriptionAdapterModel.Status), false) => dataSource.OrderBy(x => x.Status).ThenBy(x => x.Id),
                (nameof(MediaTranscriptionAdapterModel.FileSize), true) => dataSource.OrderByDescending(x => x.FileSize).ThenByDescending(x => x.Id),
                (nameof(MediaTranscriptionAdapterModel.FileSize), false) => dataSource.OrderBy(x => x.FileSize).ThenBy(x => x.Id),
                (nameof(MediaTranscriptionAdapterModel.UploadedAt), true) => dataSource.OrderByDescending(x => x.UploadedAt).ThenByDescending(x => x.Id),
                (nameof(MediaTranscriptionAdapterModel.UploadedAt), false) => dataSource.OrderBy(x => x.UploadedAt).ThenBy(x => x.Id),
