抖音影片自動轉錄整合 Implementation Plan

Archived historical implementation plan: 本檔保留當時 agent 執行計畫與脈絡；下列步驟不再作為目前 backlog 追蹤。

Goal: 讓使用者貼抖音分享文字即自動帶入標題＋網址，且背景轉錄時自動以 playwright 助手下載抖音音訊並沿用既有轉錄管線，達成抖音→文字稿自動化。

Architecture: 在 TranscriptionProcessor.DownloadFromUrlAsync 下載步驟依網址分流：抖音→新的 DouyinAudioDownloader（C# shell out 到既有 Node 助手 tools/douyin-audio/get-douyin-audio.mjs），其餘→既有 yt-dlp MediaUrlDownloader；兩者回傳相同的 MediaDownloadResult，下游分段→轉錄→繁體流程不變。建立表單以 ShareTextParser 解析分享文字自動帶入標題與網址。

Tech Stack: .NET 10、C#、Blazor Server（AntDesign）、xUnit、Node.js + @playwright/cli、ffmpeg。

設計依據： docs/superpowers/specs/2026-06-04-抖音自動轉錄整合-design.md

版本規範： 全部任務完成後，於 Task 7 一次性將 appsettings.json 的 SystemVersion 由 0.2.26 遞增為 0.2.27 (2026/06/04)（單一功能交付＝一次建置）。

using System.Text.RegularExpressions;

namespace KnowledgeExtraction.Business.Helpers;

/// <summary>
/// 解析使用者貼上的影片「分享文字」，抽出乾淨網址與標題。
/// 標題為最佳努力擷取（可由使用者修改）：取第一個網址前的文字，去除分享前綴雜訊
/// （數字代碼、時間、日期等非中日韓字 token）與「…看看【作者的作品】」包裝。
/// </summary>
public static class ShareTextParser
{
    private static readonly Regex CjkToken = new(@"[一-鿿]", RegexOptions.Compiled);
    private static readonly Regex UrlStart = new(@"https?://", RegexOptions.IgnoreCase | RegexOptions.Compiled);
    private static readonly Regex AuthorWrapper = new(@"看看【[^】]*】", RegexOptions.Compiled);

    /// <summary>抽出文字中第一個 http(s) 網址（沿用既有 UrlExtractor）。</summary>
    public static string? ExtractUrl(string? text) => UrlExtractor.ExtractFirstUrl(text);

    /// <summary>抽出標題；無法判定時回傳 null。</summary>
    public static string? ExtractTitle(string? text)
    {
        if (string.IsNullOrWhiteSpace(text))
        {
            return null;
        }

        // 取第一個網址之前的文字
        var urlMatch = UrlStart.Match(text);
        var pre = urlMatch.Success ? text[..urlMatch.Index] : text;

        // 去掉「…看看【作者的作品】」包裝，保留其後標題
        var wrap = AuthorWrapper.Match(pre);
        if (wrap.Success)
        {
            pre = pre[(wrap.Index + wrap.Length)..];
        }

        // 去掉開頭不含中日韓字的雜訊 token（代碼、時間、日期）
        var tokens = pre.Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries);
        var kept = tokens.SkipWhile(t => !CjkToken.IsMatch(t)).ToArray();
        var title = string.Join(' ', kept).Trim();
        return string.IsNullOrWhiteSpace(title) ? null : title;
    }
}

using System.Diagnostics;
using System.Text.RegularExpressions;

namespace KnowledgeExtraction.Business.Services.Transcription;

/// <summary>
/// 抖音音訊下載器：抖音 web API 需 JS 即時簽章，yt-dlp 無法下載，故以 Node + playwright 助手
/// （tools/douyin-audio/get-douyin-audio.mjs）用已登入瀏覽器攔截音訊串流並轉 mp3。
/// 僅適用於有桌面的本機環境（助手需可見瀏覽器與偶爾人工驗證）。
/// </summary>
public class DouyinAudioDownloader
{
    private readonly string nodePath;
    private readonly string scriptPath;
    private readonly int timeoutSeconds;

    public DouyinAudioDownloader(string nodePath, string scriptPath, int timeoutSeconds)
    {
        this.nodePath = string.IsNullOrWhiteSpace(nodePath) ? "node" : nodePath;
        this.scriptPath = scriptPath?.Trim() ?? string.Empty;
        this.timeoutSeconds = timeoutSeconds > 0 ? timeoutSeconds : 150;
    }

    /// <summary>判斷網址是否為抖音（含短連結 v.douyin.com）。</summary>
    public static bool IsDouyin(string? url)
    {
        if (!Uri.TryCreate(url, UriKind.Absolute, out var uri))
        {
            return false;
        }
        var host = uri.Host.ToLowerInvariant();
        return host == "douyin.com"
            || host == "v.douyin.com"
            || host.EndsWith(".douyin.com", StringComparison.Ordinal)
            || host.EndsWith("iesdouyin.com", StringComparison.Ordinal);
    }

    /// <summary>以助手腳本下載抖音音訊到 mediaRootPath，輸出 {yyyy}/{guid}.mp3。</summary>
    public async Task<MediaDownloadResult> DownloadAudioAsync(
        string url, string mediaRootPath, DateTime uploadedAt, CancellationToken cancellationToken)
    {
        if (string.IsNullOrWhiteSpace(scriptPath) || !File.Exists(scriptPath))
        {
            throw new InvalidOperationException(
                "抖音下載需設定 OpenAISettings:DouyinHelperScript 指向 get-douyin-audio.mjs，且檔案需存在。");
        }

        var year = uploadedAt.Year.ToString("0000");
        var id = Guid.NewGuid().ToString("N");
        var targetDir = Path.Combine(mediaRootPath, year);
        Directory.CreateDirectory(targetDir);
        var outPath = Path.Combine(targetDir, $"{id}.mp3");

        var arguments = $"\"{scriptPath}\" \"{url}\" --out \"{outPath}\"";
        var (stdout, stderr, exitCode) = await RunAsync(arguments, cancellationToken);
        if (exitCode != 0 || !File.Exists(outPath))
        {
            var detail = string.IsNullOrWhiteSpace(stderr) ? stdout : stderr;
            throw new InvalidOperationException($"抖音助手執行失敗 (ExitCode={exitCode})：{detail}");
        }

        var titleMatch = Regex.Match(stdout, @"^TITLE=(.*)$", RegexOptions.Multiline);
        var title = titleMatch.Success ? titleMatch.Groups[1].Value.Trim() : id;
        var safeTitle = SanitizeFileName(string.IsNullOrWhiteSpace(title) ? id : title);

        var fileInfo = new FileInfo(outPath);
        return new MediaDownloadResult(
            outPath,
            $"{year}/{id}.mp3",
            $"{safeTitle}.mp3",
            "audio/mpeg",
            fileInfo.Length);
    }

    private async Task<(string Stdout, string Stderr, int ExitCode)> RunAsync(string arguments, CancellationToken cancellationToken)
    {
        var psi = new ProcessStartInfo
        {
            FileName = nodePath,
            Arguments = arguments,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true,
        };

        using var process = Process.Start(psi)
            ?? throw new InvalidOperationException("無法啟動 Node，請確認已安裝並設定 OpenAISettings:NodePath。");

        using var timeoutCts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        timeoutCts.CancelAfter(TimeSpan.FromSeconds(timeoutSeconds));

        var stdoutTask = process.StandardOutput.ReadToEndAsync(timeoutCts.Token);
        var stderrTask = process.StandardError.ReadToEndAsync(timeoutCts.Token);
        try
        {
            await process.WaitForExitAsync(timeoutCts.Token);
        }
        catch (OperationCanceledException) when (!cancellationToken.IsCancellationRequested)
        {
            try { process.Kill(entireProcessTree: true); } catch { /* ignore */ }
            throw new InvalidOperationException(
                $"抖音助手逾時（{timeoutSeconds} 秒）。若出現驗證頁，請於瀏覽器視窗手動通過後重試。");
        }

        var stdout = await stdoutTask;
        var stderr = await stderrTask;
        return (stdout, stderr, process.ExitCode);
    }

    private static string SanitizeFileName(string name)
    {
        var invalid = Path.GetInvalidFileNameChars();
        var cleaned = new string(name.Select(c => invalid.Contains(c) ? '_' : c).ToArray()).Trim();
        return string.IsNullOrWhiteSpace(cleaned) ? "douyin_audio" : cleaned;
    }
}

    private void OnSourceUrlBlur(FocusEventArgs _)
    {
        var raw = CurrentRecord.SourceUrl;
        if (string.IsNullOrWhiteSpace(raw))
        {
            return;
        }

        var url = ShareTextParser.ExtractUrl(raw);
        if (!string.IsNullOrWhiteSpace(url))
        {
            CurrentRecord.SourceUrl = url;
        }

        if (string.IsNullOrWhiteSpace(CurrentRecord.Title))
        {
            var title = ShareTextParser.ExtractTitle(raw);
            if (!string.IsNullOrWhiteSpace(title))
            {
                CurrentRecord.Title = title;
            }
        }
    }
